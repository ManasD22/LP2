def is_safe(vertex, color, graph, colors_assigned):
    for neighbor in graph[vertex]:
        if colors_assigned[neighbor] == color:
            return False
    return True

def branch_and_bound(graph, m, colors_assigned, vertex):
    n = len(graph)

    if vertex == n:
        return True

    for color in range(1, m + 1):
        if is_safe(vertex, color, graph, colors_assigned):
            colors_assigned[vertex] = color

            if branch_and_bound(graph, m, colors_assigned, vertex + 1):
                return True

            # Backtrack
            colors_assigned[vertex] = 0

    return False

def main():
    print("Graph Coloring using Branch and Bound Algorithm")

    n = int(input("Enter the number of vertices: "))
    e = int(input("Enter the number of edges: "))
    m = int(input("Enter the number of colors: "))

    graph = [[] for _ in range(n)]

    print("Enter the edges (as pairs of vertices, 0-indexed):")
    for _ in range(e):
        u, v = map(int, input().split())
        graph[u].append(v)
        graph[v].append(u)

    colors_assigned = [0] * n

    if branch_and_bound(graph, m, colors_assigned, 0):
        print("Solution found!")
        for i, color in enumerate(colors_assigned):
            print(f"Vertex {i} --> Color {color}")
    else:
        print("No solution exists with the given number of colors.")

if __name__ == "__main__":
    main()
