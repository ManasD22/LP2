class DisjointSet:
    def __init__(self, num_vertices):
        """Initialize a disjoint set to represent the graph's vertices."""
        self.parent = list(range(num_vertices))  # Parent array representing each vertex's parent (initially, each vertex is its own parent)

    def find_parent(self, vertex):
        """Find the root parent of a vertex using path compression."""
        if self.parent[vertex] != vertex:
            self.parent[vertex] = self.find_parent(self.parent[vertex])  # Recursively find the parent and apply path compression
        return self.parent[vertex]

    def union_sets(self, vertex1, vertex2):
        """Union two sets by connecting their roots."""
        root1 = self.find_parent(vertex1)  # Find root of vertex1
        root2 = self.find_parent(vertex2)  # Find root of vertex2
        if root1 != root2:  # If the roots are different, perform union
            self.parent[root2] = root1  # Attach root2's tree under root1

def kruskal_mst(num_vertices, edge_list):
    """Kruskal's algorithm to find the minimum spanning tree (MST) of a graph."""
    
    disjoint_set = DisjointSet(num_vertices)  # Initialize disjoint set to keep track of connected components
    sorted_edges = sorted(edge_list, key=lambda edge: edge[2])  # Sort edges based on their weights in ascending order
    # key=lambda edge sorts the edges in ascending order based on weight
    # edge[2] is accessing the third element (weight) of each edge tuple
    
    mst_edges = []  # List to store edges in the MST
    total_weight = 0  # Variable to keep track of the total weight of the MST

    # Step through each edge in sorted order
    for vertex1, vertex2, weight in sorted_edges:
        # If vertex1 and vertex2 belong to different sets, adding the edge won't form a cycle
        if disjoint_set.find_parent(vertex1) != disjoint_set.find_parent(vertex2):
            disjoint_set.union_sets(vertex1, vertex2)  # Union the sets of vertex1 and vertex2
            mst_edges.append((vertex1, vertex2, weight))  # Add the edge to MST
            total_weight += weight  # Add the weight of the edge to the total weight

    return mst_edges, total_weight

# Input from user
num_vertices = int(input("Enter number of vertices: "))  # Number of vertices in the graph
num_edges = int(input("Enter number of edges: "))  # Number of edges in the graph
edges = []  # List to store edges

# Read edges from user input
for _ in range(num_edges):
    u, v, w = map(int, input("Enter edge (start_vertex end_vertex weight): ").split())  # Accept each edge and its weight
    edges.append((u, v, w))  # Add the edge as a tuple (start_vertex, end_vertex, weight)

# Run Kruskal's algorithm to get the MST
mst_edges, mst_total_weight = kruskal_mst(num_vertices, edges)

# Output the results
print("\nEdges in Minimum Spanning Tree:")
for u, v, w in mst_edges:
    print(f"{u} - {v} : {w}")  # Print each edge in the MST
print("Total weight of MST:", mst_total_weight)  # Print the total weight of the MST
