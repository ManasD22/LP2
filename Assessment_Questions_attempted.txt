1. Write a trigger on account that updates all related contacts phone numbers when the account phone is changed.

trigger AccountPhoneTrigger on Account (after update) {
    AccountPhoneTriggerHandler.handleAfterUpdate(Trigger.new, Trigger.oldMap);
}

public class AccountPhoneTriggerHandler {
    
    public static void handleAfterUpdate(List<Account> newAccList, Map<Id, Account> oldAccMap) {
        
        Set<Id> changedPhoneAccIds = new Set<Id>();
        
        for (Account newAcc : newAccList) {
            Account oldAcc = oldAccMap.get(newAcc.Id);
            
            if (newAcc.Phone != oldAcc.Phone) {
                changedPhoneAccIds.add(newAcc.Id);
            }
        }
        
        if (changedPhoneAccIds.size() > 0) {
            List<Contact> contactsToUpdate = [SELECT Id, Phone, AccountId 
                                              FROM Contact 
                                              WHERE AccountId IN :changedPhoneAccIds];
            
            Map<Id, String> accIdPhoneMap = new Map<Id, String>();
            for (Account newAcc : newAccList) {
                if (changedPhoneAccIds.contains(newAcc.Id)) {
                    accIdPhoneMap.put(newAcc.Id, newAcc.Phone);
                }
            }
            
            for (Contact con : contactsToUpdate) {
                con.Phone = accIdPhoneMap.get(con.AccountId);
            }
            
            if (contactsToUpdate.size() > 0) {
                update contactsToUpdate;
            }
        }
    }
}



2. write a trigger on opportunity that updates all related cases to closed when opportunity stagename = closedwon

trigger OpportunityClosedTrigger on Opportunity (after update) {
    OpportunityClosedTriggerHandler.handleAfterUpdate(Trigger.new, Trigger.oldMap);
}

public class OpportunityClosedTriggerHandler {
    
    public static void handleAfterUpdate(List<Opportunity> newOppList, Map<Id, Opportunity> oldOppMap) {
        
        Set<Id> closedWonAccIds = new Set<Id>();
        
        for (Opportunity newOpp : newOppList) {
            Opportunity oldOpp = oldOppMap.get(newOpp.Id);
            
            if (newOpp.StageName == 'Closed Won' && oldOpp.StageName != 'Closed Won') {
                if (newOpp.AccountId != null) {
                    closedWonAccIds.add(newOpp.AccountId);
                }
            }
        }
        
        if (closedWonAccIds.size() > 0) {
            List<Case> casesToUpdate = [SELECT Id, Status, AccountId 
                                        FROM Case 
                                        WHERE AccountId IN :closedWonAccIds
                                        AND Status != 'Closed'];
            
            for (Case c : casesToUpdate) {
                c.Status = 'Closed';
            }
            
            if (casesToUpdate.size() > 0) {
                update casesToUpdate;
            }
        }
    }
}



3. Write a trigger on account that prevents deletion if the account has related opportunities.

trigger AccountDeletionTrigger on Account (before delete) {
    AccountDeletionTriggerHandler.handleBeforeDelete(Trigger.old, Trigger.oldMap);
}

public class AccountDeletionTriggerHandler {
    
    public static void handleBeforeDelete(List<Account> oldAccList, Map<Id, Account> oldAccMap) {
        
        Set<Id> accIds = oldAccMap.keySet();
        
        List<Opportunity> relatedOpps = [SELECT Id, Name, AccountId 
                                         FROM Opportunity 
                                         WHERE AccountId IN :accIds];
        
        Map<Id, Integer> accOppCountMap = new Map<Id, Integer>();
        
        for (Opportunity opp : relatedOpps) {
            if (!accOppCountMap.containsKey(opp.AccountId)) {
                accOppCountMap.put(opp.AccountId, 0);
            }
            accOppCountMap.put(opp.AccountId, accOppCountMap.get(opp.AccountId) + 1);
        }
        
        for (Account acc : oldAccList) {
            if (accOppCountMap.containsKey(acc.Id)) {
                acc.addError('Cannot delete Account it has Opportunities related to it');
            }
        }
    }
}

4. write a trigger on case that updates the parent accounts rating to 'hot' if more than 3 cases are created for that account

trigger CaseAccountRatingTrigger on Case (after insert) {
    CaseAccountRatingTriggerHandler.handleAfterInsert(Trigger.new);
}

public class CaseAccountRatingTriggerHandler {
    
    public static void handleAfterInsert(List<Case> newCaseList) {
        
        Set<Id> accIds = new Set<Id>();
        
        for (Case c : newCaseList) {
            if (c.AccountId != null) {
                accIds.add(c.AccountId);
            }
        }
        
        if (accIds.size() == 0) return;
        
        List<AggregateResult> caseCountResults = [SELECT AccountId, COUNT(Id) total 
                                                   FROM Case 
                                                   WHERE AccountId IN :accIds
                                                   GROUP BY AccountId];
        
        Map<Id, Integer> accCaseCountMap = new Map<Id, Integer>();
        
        for (AggregateResult ar : caseCountResults) {
            accCaseCountMap.put((Id) ar.get('AccountId'), (Integer) ar.get('total'));
        }
        
        List<Account> accountsToUpdate = [SELECT Id, Name, Rating 
                                           FROM Account 
                                           WHERE Id IN :accIds];
        
        List<Account> accToUpdate = new List<Account>();
        
        for (Account acc : accountsToUpdate) {
            if (accCaseCountMap.containsKey(acc.Id) && 
                accCaseCountMap.get(acc.Id) > 3 && 
                acc.Rating != 'Hot') {
                    acc.Rating = 'Hot';
                    accToUpdate.add(acc);
            }
        }
        
        if (accToUpdate.size() > 0) {
            update accToUpdate;
        }
    }
}

1. Write a batch apex class that updates a custom field Email_Domain__c on contacts with the domain part of their Email address.

global class ContactEmailDomainBatch implements Database.Batchable<sObject> {
    
    global Database.QueryLocator start(Database.BatchableContext bc) {
        return Database.getQueryLocator([
            SELECT Id, Email, Email_Domain__c 
            FROM Contact 
            WHERE Email != null
        ]);
    }
    
    global void execute(Database.BatchableContext bc, List<Contact> contactList) {
        
        List<Contact> contactsToUpdate = new List<Contact>();
        
        for (Contact con : contactList) {
            if (con.Email != null && con.Email.contains('@')) {
                
                String domain = con.Email.substringAfter('@');
                
                if (con.Email_Domain__c != domain) {
                    con.Email_Domain__c = domain;
                    contactsToUpdate.add(con);
                }
            }
        }
        
        if (contactsToUpdate.size() > 0) {
            update contactsToUpdate;
        }
    }
    
    global void finish(Database.BatchableContext bc) {
    }
}

2. write a batch apex class that updates account rating to 'Hot' if annual revenue > 1000000 and warm if between 500000 to 1000000

global class AccountRatingBatch implements Database.Batchable<sObject> {
    
    global Database.QueryLocator start(Database.BatchableContext bc) {
        return Database.getQueryLocator([
            SELECT Id, Name, AnnualRevenue, Rating 
            FROM Account 
            WHERE AnnualRevenue != null
        ]);
    }
    
    global void execute(Database.BatchableContext bc, List<Account> accountList) {
        List<Account> accountsToUpdate = new List<Account>();
        
        for (Account acc : accountList) {
            if (acc.AnnualRevenue > 1000000 && acc.Rating != 'Hot') {
                acc.Rating = 'Hot';
                accountsToUpdate.add(acc);
            } else if (acc.AnnualRevenue >= 500000 && acc.AnnualRevenue <= 1000000 && acc.Rating != 'Warm') {
                acc.Rating = 'Warm';
                accountsToUpdate.add(acc);
            }
        }
        
        if (accountsToUpdate.size() > 0) {
            update accountsToUpdate;
        }
    }
    
    global void finish(Database.BatchableContext bc) {}
}
